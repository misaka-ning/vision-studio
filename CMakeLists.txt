cmake_minimum_required(VERSION 3.21)
project(VisionStudio VERSION 2.0.0 LANGUAGES CXX)
set(VISION_STUDIO_APP_VERSION "2.0.0-beta.2")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(VISION_RESOURCE_FALLBACK "${CMAKE_SOURCE_DIR}" CACHE STRING "Resource root used when no adjacent resources exist")
include(CTest)
if(NOT CMAKE_PREFIX_PATH AND EXISTS "$ENV{HOME}/Qt/6.8.3/gcc_64")
    list(APPEND CMAKE_PREFIX_PATH "$ENV{HOME}/Qt/6.8.3/gcc_64")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Svg Network WebChannel WebEngineWidgets)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
endif()
find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc imgcodecs dnn videoio)
add_library(vision-core STATIC src/core/visionengine.cpp src/core/inferenceworker.cpp src/core/ptbackend.cpp src/core/videorecorder.cpp src/core/gpuruntime.cpp src/core/onnxcudabackend.cpp)
target_compile_definitions(vision-core PRIVATE VISION_PROJECT_DIR="${VISION_RESOURCE_FALLBACK}")
target_include_directories(vision-core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision-core PUBLIC Qt6::Core Qt6::Gui ${OpenCV_LIBS})
add_executable(vision-studio src/main.cpp src/ui/mainwindow.cpp src/ui/imagecanvas.cpp src/ui/modelviewer.cpp src/ui/icons.cpp resources.qrc)
target_link_libraries(vision-studio PRIVATE vision-core Qt6::Widgets Qt6::Svg Qt6::Network Qt6::WebChannel Qt6::WebEngineWidgets)
target_compile_definitions(vision-studio PRIVATE VISION_PROJECT_DIR="${VISION_RESOURCE_FALLBACK}" VISION_STUDIO_APP_VERSION="${VISION_STUDIO_APP_VERSION}")
set_target_properties(vision-studio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_options(vision-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(vision-studio PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING AND EXISTS "${CMAKE_SOURCE_DIR}/tests/core_tests.cpp")
    add_executable(vision-core-tests tests/core_tests.cpp)
    target_link_libraries(vision-core-tests PRIVATE vision-core Qt6::Test)
    target_compile_definitions(vision-core-tests PRIVATE VISION_TEST_DIR="${CMAKE_SOURCE_DIR}/tests")
    add_test(NAME vision-core-tests COMMAND vision-core-tests)
    set_tests_properties(vision-core-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(EXISTS "${CMAKE_SOURCE_DIR}/tests/recording_tests.cpp")
        add_executable(vision-recording-tests tests/recording_tests.cpp)
        target_link_libraries(vision-recording-tests PRIVATE vision-core Qt6::Test)
        target_compile_definitions(vision-recording-tests PRIVATE VISION_PROJECT_DIR="${CMAKE_SOURCE_DIR}")
        add_test(NAME vision-recording-tests COMMAND vision-recording-tests)
        set_tests_properties(vision-recording-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
    endif()
    if(EXISTS "${CMAKE_SOURCE_DIR}/tests/pt_backend_tests.cpp")
        add_executable(vision-pt-tests tests/pt_backend_tests.cpp)
        target_link_libraries(vision-pt-tests PRIVATE vision-core Qt6::Test)
        target_compile_definitions(vision-pt-tests PRIVATE VISION_PROJECT_DIR="${CMAKE_SOURCE_DIR}")
        add_test(NAME vision-pt-tests COMMAND vision-pt-tests)
        set_tests_properties(vision-pt-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
    endif()
    if(EXISTS "${CMAKE_SOURCE_DIR}/tests/gpu_backend_tests.cpp")
        add_executable(vision-gpu-tests tests/gpu_backend_tests.cpp)
        target_link_libraries(vision-gpu-tests PRIVATE vision-core Qt6::Test)
        target_compile_definitions(vision-gpu-tests PRIVATE VISION_PROJECT_DIR="${CMAKE_SOURCE_DIR}")
        add_test(NAME vision-gpu-tests COMMAND vision-gpu-tests)
        set_tests_properties(vision-gpu-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 600 LABELS "gpu")
    endif()
    if(EXISTS "${CMAKE_SOURCE_DIR}/tests/ui_tests.cpp")
        add_executable(vision-ui-tests tests/ui_tests.cpp src/ui/mainwindow.cpp src/ui/imagecanvas.cpp src/ui/modelviewer.cpp src/ui/icons.cpp resources.qrc)
        target_link_libraries(vision-ui-tests PRIVATE vision-core Qt6::Widgets Qt6::Svg Qt6::Test Qt6::Network Qt6::WebChannel Qt6::WebEngineWidgets)
        target_compile_definitions(vision-ui-tests PRIVATE VISION_PROJECT_DIR="${CMAKE_SOURCE_DIR}")
        add_test(NAME vision-ui-tests COMMAND vision-ui-tests)
        set_tests_properties(vision-ui-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    endif()
    if(EXISTS "${CMAKE_SOURCE_DIR}/tests/modelviewer_tests.cpp")
        add_executable(vision-modelviewer-tests tests/modelviewer_tests.cpp src/ui/modelviewer.cpp)
        target_link_libraries(vision-modelviewer-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::WebChannel Qt6::WebEngineWidgets Qt6::Test)
        target_include_directories(vision-modelviewer-tests PRIVATE src)
        target_compile_definitions(vision-modelviewer-tests PRIVATE VISION_PROJECT_DIR="${CMAKE_SOURCE_DIR}")
        add_test(NAME vision-modelviewer-tests COMMAND vision-modelviewer-tests)
        set_tests_properties(vision-modelviewer-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QTWEBENGINE_CHROMIUM_FLAGS=--disable-gpu" TIMEOUT 180)
    endif()
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME vision-netron-server-tests COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/netron_server_tests.py")
        set_tests_properties(vision-netron-server-tests PROPERTIES TIMEOUT 45)
        if(EXISTS "${CMAKE_SOURCE_DIR}/tests/gpu_runtime_tests.py")
            add_test(NAME vision-gpu-runtime-tests COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/gpu_runtime_tests.py")
            set_tests_properties(vision-gpu-runtime-tests PROPERTIES TIMEOUT 60)
        endif()
    endif()
endif()
install(TARGETS vision-studio RUNTIME DESTINATION bin)
