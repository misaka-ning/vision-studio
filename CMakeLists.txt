        foreach(_vision_eval_python IN ITEMS
                "${CMAKE_SOURCE_DIR}/runtime/bin/python" "${CMAKE_SOURCE_DIR}/runtime/python.exe"
                "$ENV{VISION_STUDIO_HOME}/runtime/bin/python" "$ENV{VISION_STUDIO_HOME}/runtime/python.exe"
                "/opt/VisionStudio/runtime/bin/python" "${Python3_EXECUTABLE}")
            if(EXISTS "${_vision_eval_python}" AND NOT IS_DIRECTORY "${_vision_eval_python}")
                set(VISION_STUDIO_TEST_PYTHON "${_vision_eval_python}" CACHE FILEPATH
                    "Python interpreter with the Vision Studio CPU helper dependencies for evaluation tests" FORCE)
                break()
            endif()
        endforeach()
    endif()
    if(VISION_STUDIO_TEST_PYTHON)
        add_test(NAME vision-evaluation-python-tests COMMAND "${VISION_STUDIO_TEST_PYTHON}" -B "${CMAKE_SOURCE_DIR}/tests/evaluation_tests.py")
        add_test(NAME vision-evaluation-reporting-tests COMMAND "${VISION_STUDIO_TEST_PYTHON}" -B "${CMAKE_SOURCE_DIR}/tests/evaluation_reporting_tests.py")

            OUTPUT_QUIET TIMEOUT 15)
        if(NOT _vision_eval_python_status STREQUAL "0")
            message(WARNING "Evaluation Python dependency check failed; the test remains enabled and must pass for release QA. Set VISION_STUDIO_TEST_PYTHON to a prepared runtime. ${_vision_eval_python_error}")
        endif()
    else()
        message(WARNING "No Python interpreter for evaluation tests. Set VISION_STUDIO_TEST_PYTHON to a prepared runtime; the registered test will fail until one is available.")
        add_test(NAME vision-evaluation-python-tests COMMAND "${CMAKE_COMMAND}" -E false)
    endif()
    set_tests_properties(vision-evaluation-python-tests PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 240)
